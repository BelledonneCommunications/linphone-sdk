cmake_minimum_required(VERSION 3.22)

project(cs-wrapper)

#No hints because we don't want to suppose the used version.
find_program(MSBUILD_EXECUTABLE
		NAMES MSBuild.exe msbuild
)

if(NOT MSBUILD_EXECUTABLE)
	message(FATAL_ERROR "MSBuild.exe not found! You may call vcvarsall.bat to set the developer environment")
endif()

#avoid to have conflicts between files generated by cmake and dotnet restore
set(WORKING_BUILD_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/work")

configure_file(${LINPHONESDK_INSTALL_DIR}/share/linphonecs/LinphoneWrapper.cs ${WORKING_BUILD_DIRECTORY}/LinphoneWrapper.cs COPYONLY)
configure_file(${LINPHONESDK_DIR}/cmake/Windows/wrapper/Cs${WINDOWS_VARIANT}Wrapper.csproj.in ${WORKING_BUILD_DIRECTORY}/Cs${WINDOWS_VARIANT}Wrapper.csproj @ONLY)
configure_file(${LINPHONESDK_DIR}/cmake/Windows/wrapper/Properties/AssemblyInfo.cs ${WORKING_BUILD_DIRECTORY}/Properties/AssemblyInfo.cs COPYONLY)
configure_file(${LINPHONESDK_DIR}/cmake/Windows/wrapper/Properties/CsWrapper.rd.xml ${WORKING_BUILD_DIRECTORY}/Properties/CsWrapper.rd.xml COPYONLY)

message(STATUS "MSBuild: ${MSBUILD_EXECUTABLE}")
message(STATUS "INSTALLATION: ${LINPHONESDK_INSTALL_DIR}")
message(STATUS "LINPHONESDK_DIR: ${LINPHONESDK_DIR}")
message(STATUS "LINPHONE_PLATFORM: ${LINPHONE_PLATFORM}")
message(STATUS "BUILD_TYPE: ${BUILD_TYPE}")
message(STATUS "WINDOWS_VARIANT: ${WINDOWS_VARIANT}")
message(STATUS "WORKING_DIRECTORY: ${WORKING_BUILD_DIRECTORY}")

message("Command : ${MSBUILD_EXECUTABLE} Cs${WINDOWS_VARIANT}Wrapper.csproj /p:BaseOutputPath=${LINPHONESDK_INSTALL_DIR} /p:MDILCompile=true /t:build /p:Configuration=${BUILD_TYPE} /p:Platform=${LINPHONE_PLATFORM}")

# Check file script to avoid silent error from msbuild
set(SCRIPT "${WORKING_BUILD_DIRECTORY}/check_wrapper_dll.cmake")
file(GENERATE OUTPUT "${SCRIPT}"
	CONTENT
		"if(NOT EXISTS [==[${LINPHONESDK_INSTALL_DIR}/bin/CsWrapper.dll]==])\nmessage(FATAL_ERROR [==[File not found: ${LINPHONESDK_INSTALL_DIR}/bin/CsWrapper.dll]==])\nendif()"
)
###################

set(DLL_FILE ${LINPHONESDK_INSTALL_DIR}/bin/CsWrapper.dll)

add_custom_command(
	OUTPUT "${DLL_FILE}"
	COMMAND ${CMAKE_COMMAND} -E echo "Restoring wrapper project"
	COMMAND "${MSBUILD_EXECUTABLE}" "Cs${WINDOWS_VARIANT}Wrapper.csproj" "-t:restore"
	COMMAND ${CMAKE_COMMAND} -E echo "Building wrapper : msbuild Cs${WINDOWS_VARIANT}Wrapper.csproj /p:BaseOutputPath=${LINPHONESDK_INSTALL_DIR} /p:MDILCompile=true /t:build /p:Configuration=${BUILD_TYPE} /p:Platform=${LINPHONE_PLATFORM}"
	COMMAND "${MSBUILD_EXECUTABLE}" "Cs${WINDOWS_VARIANT}Wrapper.csproj" "/p:BaseOutputPath=${LINPHONESDK_INSTALL_DIR}" "/p:MDILCompile=true" "/t:Rebuild" "/p:Configuration=${BUILD_TYPE}" "/p:Platform=${LINPHONE_PLATFORM}"
	COMMAND ${CMAKE_COMMAND} -P "${SCRIPT}"
	WORKING_DIRECTORY ${WORKING_BUILD_DIRECTORY}
	USES_TERMINAL
	DEPENDS ${LINPHONESDK_INSTALL_DIR}/share/linphonecs/LinphoneWrapper.cs
		${LINPHONESDK_DIR}/cmake/Windows/wrapper/Cs${WINDOWS_VARIANT}Wrapper.csproj.in
		${LINPHONESDK_DIR}/cmake/Windows/wrapper/Properties/AssemblyInfo.cs
		${LINPHONESDK_DIR}/cmake/Windows/wrapper/Properties/CsWrapper.rd.xml
)

add_custom_target(cs-wrapper-target ALL
	DEPENDS "${DLL_FILE}"
)
